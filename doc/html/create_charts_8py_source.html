11def crea_cartelle_e_sposta_file(directory):
12 # Controlla se la directory esiste
13 if not os.path.exists(directory):
14 print(f"La directory {directory} non esiste.")
15 return
16
17 # Scorre tutti i file nella directory
18 for file in os.listdir(directory):
19 # Controlla se il file è un file .csv
20 if file.startswith("result-") and file.endswith(".csv"):
21 # Estrai il suffisso <DATA>-<ORA> dal nome del file
22 nome_base = file[7:-4] # Rimuove "result-" all'inizio e ".csv" alla fine
23 file_txt = f"description-{nome_base}.txt"
24
25 # Controlla se il file .txt corrispondente esiste
26 if os.path.exists(os.path.join(directory, file_txt)):
27 # Crea una cartella con il nome result-<DATA>-<ORA>
28 nome_cartella = f"result-{nome_base}"
29 percorso_cartella = os.path.join(directory, nome_cartella)
30
31 # Crea la cartella se non esiste già
32 if not os.path.exists(percorso_cartella):
33 os.makedirs(percorso_cartella)
34 print(f"Creata cartella: {percorso_cartella}")
35
36 # Sposta il file .csv nella cartella
37 shutil.move(os.path.join(directory, file), os.path.join(percorso_cartella, file))
38 print(f"Spostato {file} in {percorso_cartella}")
39
40 # Sposta il file .txt nella cartella
41 shutil.move(os.path.join(directory, file_txt), os.path.join(percorso_cartella, file_txt))
42 print(f"Spostato {file_txt} in {percorso_cartella}")
43 else:
44 print(f"Il file {file_txt} non esiste per il file {file}")
