35void populateBlockOfMatrix(float *A, int bRow, int bCol, int bs, int size){
36 for(int rowInsideBlock = 0; rowInsideBlock < bs; ++rowInsideBlock){
37 for(int colInsideBlock = 0; colInsideBlock < bs; ++colInsideBlock){
38 int item = (bRow * bs * size) + (bCol * bs) + (rowInsideBlock * size) + colInsideBlock;
39 if(rowInsideBlock * bs + colInsideBlock < bs * bs){
40 A[rowInsideBlock * bs + colInsideBlock] = item;
41 }
42 }
43 }
44}
