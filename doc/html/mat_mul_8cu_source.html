205__global__ void matrixMultiplyTensorCore(const half *a, const half *b, float *d_c, int n) {
206 const int blockSize = BLOCK_SIZE * BLOCK_SIZE;
207
208 __shared__ half As [SHARED_PAGE_COUNT * blockSize];
209 __shared__ half Bs [SHARED_PAGE_COUNT * blockSize];
210 // __shared__ float Cs [BLOCK_SIZE * BLOCK_SIZE];
211 // __shared__ float Acc [BLOCK_SIZE * BLOCK_SIZE];
212 extern __shared__ float sharedMem[];
213 float *Cs = sharedMem; // La matrice Cs è allocata in shared memory
214 float *Acc = sharedMem + 2 * (blockSize); // La matrice Acc è allocata in shared memory
215
216 const int numPages = n / (SHARED_PAGE_COUNT * 32) || 1; // 32 è il numero di elementi per pagina di shared memory
217 const int numBlocks = numPages < 2 ? n / BLOCK_SIZE : SHARED_PAGE_COUNT; // Se la matrice è piccola non tutti i blocchi di shared memory sono utilizzati
218 const int blockRow = blockIdx.y;
219 const int blockCol = blockIdx.x;
220
221
222 // __syncthreads();
223 // Inizializza la matrice Acc a zero
224 clearBlockToShared(Cs);
225 clearBlockToShared(Cs + blockSize);
226 clearBlockToShared(Acc);
227 // Ciclo sulle pagine
228 for(int p = 0; p < numPages; ++p){
229 // Carica la riga di blocchi della matrice A in shared memory
230 const int pageOffset = p * SHARED_PAGE_COUNT; // Calcola l'offset della pagina in numero di blocchi
231
232 // Carica il blocco dalla matrice C in shared memory
233 loadBlockToShared(d_c, Cs, blockRow, blockCol, n);
234 __syncthreads();
235 // Ciclo all'interno della pagina
236 for(int k = 0; k < numBlocks; ++k){
237
238 // Carica i blocchi in shared memory
239 loadBlockToShared(a, As, blockRow, pageOffset + k, n);
240 loadBlockToShared(b, Bs, pageOffset + k, blockCol, n);
241 __syncthreads(); // Attendi il caricamento dei blocchi in shared memory
242
243 // Moltiplica i blocchi
244 blockMatrixMul(As, Bs, Cs, BLOCK_SIZE);
245 __syncthreads(); // Attendi i thread dei primi 8 warp per completare la computazione
246
247 Acc[threadIdx.x] = Acc[threadIdx.x] + Cs[threadIdx.x] + Cs[threadIdx.x + blockSize];
248 __syncthreads(); // Attendi il completamento della somma
249
250 clearBlockToShared(Cs);
251 __syncthreads();
252 }
253
254 copyBlockToGlobal(Acc, d_c, blockRow, blockCol, n);
255 __syncthreads();
256
257 }
258}
